cmake_minimum_required(VERSION 3.16)
project(kv260_gemm_runtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(protocol_test test_runtime.cpp)
target_include_directories(protocol_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME wire5_protocol_and_shapes COMMAND protocol_test)

option(BUILD_XRT_OWNER "Build exact qualified v5 owner/plugin on AArch64 Linux; never load PL" OFF)
if(BUILD_XRT_OWNER)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    message(FATAL_ERROR "XRT owner builds require explicit AArch64 Linux SDK; no Windows/WSL fallback")
  endif()
  find_path(XRT_INCLUDE_DIR xrt/xrt_device.h PATHS /usr/include REQUIRED)
  find_library(XRT_COREUTIL_LIBRARY NAMES xrt_coreutil REQUIRED)
  find_library(UUID_LIBRARY NAMES uuid libuuid.so.1 REQUIRED)
  find_library(CRYPTO_LIBRARY NAMES crypto libcrypto.so.3 REQUIRED)
  find_package(Threads REQUIRED)
  add_executable(native_owner owner.cpp model_gemm_runtime.cpp)
  target_include_directories(native_owner PRIVATE ${XRT_INCLUDE_DIR})
  target_compile_options(native_owner PRIVATE -O2 -Wall -Wextra -Werror)
  target_link_libraries(native_owner PRIVATE ${XRT_COREUTIL_LIBRARY} ${UUID_LIBRARY} ${CRYPTO_LIBRARY} Threads::Threads)
  add_library(quant_v5 SHARED ../../model/v5/b_plugin.cpp ../../model/v5/client.cpp)
  target_include_directories(quant_v5 PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ../../model/v5 ../../model/vendor/onnxruntime)
  target_compile_options(quant_v5 PRIVATE -O2 -Wall -Wextra -Werror -fvisibility=hidden -fno-fast-math -ffp-contract=off)
  target_link_libraries(quant_v5 PRIVATE Threads::Threads)
endif()
